<%*
// Templater 模板：文章写完后在该文章里插入本模板，把发布时间改成现在、去掉 published: false，
// 并把文件名开头的日期改成今天。之后用 Obsidian Git 提交并推送即可上线。
const file = tp.config.target_file;
const now = tp.date.now("YYYY-MM-DD HH:mm:ss ZZ"); // 必须带时区，否则会被当成 UTC，晚 8 小时上线
await app.fileManager.processFrontMatter(file, (fm) => {
  fm.date = now;
  delete fm.published;
});
await tp.file.rename(tp.date.now("YYYY-MM-DD") + tp.file.title.replace(/^\d{4}-\d{2}-\d{2}/, ""));
-%>
